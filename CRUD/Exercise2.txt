1. GPA between 3.0 and 3.5
db.students.find({
  gpa: { $gte: 3.0, $lte: 3.5 }
})

2. Students with more than 5 courses
db.students.find({
  courses: { $size: 5 }
})

For more than 5, use:

db.students.find({
  $expr: { $gt: [{ $size: "$courses" }, 5] }
})

3. Top 10 Students by GPA
db.students.find()
.sort({ gpa: -1 })
.limit(10)

4. Count Students by City
db.students.aggregate([
  { $group: { _id: "$city", count: { $sum: 1 } } }
])
