1. Average GPA by Department
db.students.aggregate([
  {
    $group: {
      _id: "$department",
      avgGPA: { $avg: "$gpa" }
    }
  }
])

2. Most Popular Courses
db.students.aggregate([
  { $unwind: "$courses" },
  {
    $group: {
      _id: "$courses",
      count: { $sum: 1 }
    }
  },
  { $sort: { count: -1 } }
])

3. Student Performance Report
db.students.aggregate([
  {
    $project: {
      name: 1,
      gpa: 1,
      performance: {
        $cond: {
          if: { $gte: ["$gpa", 3.5] },
          then: "Excellent",
          else: "Average"
        }
      }
    }
  }
])
